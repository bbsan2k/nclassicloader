cmake_minimum_required(VERSION 3.28)
if(CMAKE_HOST_APPLE AND NOT DEFINED CMAKE_OSX_DEPLOYMENT_TARGET)
  set(CMAKE_OSX_DEPLOYMENT_TARGET "13.0" CACHE STRING "Minimum macOS version")
endif()
if(WIN32 OR CMAKE_HOST_WIN32)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>" CACHE STRING "Static compiler runtime")
endif()
project(ClassicMenuPortable VERSION 0.5.0 LANGUAGES C CXX)
if(APPLE)
  enable_language(OBJCXX)
  set(CMAKE_OBJCXX_STANDARD 20)
  set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(CLOVER_BUILD_TESTS "Build portable tests" ON)
set(CLOVER_ASSET_DIR "${CMAKE_SOURCE_DIR}/.build/portable-assets" CACHE PATH "Locally prepared Clover assets")
if(NOT EXISTS "${CMAKE_SOURCE_DIR}/third_party/ares/ares/sfc/sfc.hpp" OR NOT EXISTS "${CMAKE_SOURCE_DIR}/third_party/SDL/CMakeLists.txt")
  message(FATAL_ERROR "Run python3 scripts/bootstrap-portable.py first (pinned build sources)")
endif()
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SDL_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory(third_party/SDL EXCLUDE_FROM_ALL)
if(CLOVER_BUILD_TESTS)
  enable_testing()
endif()
add_subdirectory(portable/core)
if(CLOVER_BUILD_TESTS)
  add_executable(clover_game_audit portable/core/tests/game_audit.cpp)
  target_link_libraries(clover_game_audit PRIVATE clover_ares)
  target_include_directories(clover_game_audit SYSTEM PRIVATE third_party/vendor)
endif()
set(ZLIB_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(ZLIB_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(ZLIB_BUILD_STATIC ON CACHE BOOL "" FORCE)
set(ZLIB_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory(third_party/zlib EXCLUDE_FROM_ALL)
add_library(clover_import STATIC portable/import/importer.cpp portable/import/skin.cpp portable/import/image_reader.cpp portable/import/sfrom.cpp)
target_include_directories(clover_import PUBLIC portable/import)
target_include_directories(clover_import SYSTEM PUBLIC third_party/vendor)
target_link_libraries(clover_import PRIVATE ZLIB::ZLIBSTATIC)
target_compile_features(clover_import PUBLIC cxx_std_20)
if(CLOVER_BUILD_TESTS)
  add_executable(clover_import_tests portable/import/importer_tests.cpp)
  target_link_libraries(clover_import_tests PRIVATE clover_import)
  add_test(NAME clover_import_tests COMMAND clover_import_tests)
  add_executable(clover_image_reader_tests portable/import/image_reader_tests.cpp)
  target_link_libraries(clover_image_reader_tests PRIVATE clover_import ZLIB::ZLIBSTATIC)
  add_test(NAME clover_image_reader_tests COMMAND clover_image_reader_tests)
  add_executable(clover_sfrom_tests portable/import/sfrom_tests.cpp)
  target_link_libraries(clover_sfrom_tests PRIVATE clover_import)
  add_test(NAME clover_sfrom_tests COMMAND clover_sfrom_tests)
  add_executable(clover_region_tests portable/import/region_tests.cpp)
  target_link_libraries(clover_region_tests PRIVATE clover_import)
  if(EXISTS "${CMAKE_SOURCE_DIR}/SNES-Mini-Kernels/dp-shvc-release-v2.0.12-0-gbff4fb3.tar.gz")
    add_test(NAME clover_region_tests COMMAND clover_region_tests "${CMAKE_SOURCE_DIR}/SNES-Mini-Kernels")
  endif()
  set(CLOVER_TEST_IMAGE "" CACHE FILEPATH "Optional local stock image for import/cache tests")
  if(EXISTS "${CLOVER_TEST_IMAGE}")
    add_test(NAME clover_stock_import COMMAND clover_import_tests "${CLOVER_TEST_IMAGE}")
  endif()
endif()
add_executable(ClassicMenu WIN32 MACOSX_BUNDLE portable/frontend/main.cpp)
if(APPLE)
  find_library(CLOVER_APPKIT AppKit REQUIRED)
  add_library(clover_native_menu STATIC portable/frontend/native_menu.mm)
  target_link_libraries(clover_native_menu PUBLIC SDL3::SDL3-static "${CLOVER_APPKIT}")
  target_compile_options(clover_native_menu PRIVATE -fobjc-arc)
  target_link_libraries(ClassicMenu PRIVATE clover_native_menu)
  if(CLOVER_BUILD_TESTS)
    add_executable(clover_native_menu_tests portable/frontend/native_menu_tests.mm)
    target_compile_options(clover_native_menu_tests PRIVATE -fobjc-arc)
    target_link_libraries(clover_native_menu_tests PRIVATE clover_native_menu)
    add_test(NAME clover_native_menu_tests COMMAND clover_native_menu_tests)
  endif()
endif()
if(CLOVER_BUILD_TESTS)
  add_executable(clover_geometry_tests portable/frontend/geometry_tests.cpp)
  target_link_libraries(clover_geometry_tests PRIVATE SDL3::SDL3-static)
  add_test(NAME clover_geometry_tests COMMAND clover_geometry_tests)
endif()
target_include_directories(ClassicMenu SYSTEM PRIVATE third_party/vendor)
target_include_directories(ClassicMenu PRIVATE portable/core)
target_link_libraries(ClassicMenu PRIVATE clover_ares clover_import SDL3::SDL3-static)
set_target_properties(ClassicMenu PROPERTIES MACOSX_BUNDLE_BUNDLE_NAME "Classic Menu" MACOSX_BUNDLE_GUI_IDENTIFIER "local.snesc.classic-menu.portable" MACOSX_BUNDLE_SHORT_VERSION_STRING "0.5.0" MACOSX_BUNDLE_BUNDLE_VERSION "1")
if(MSVC)
  target_compile_options(ClassicMenu PRIVATE /utf-8 /EHsc)
endif()
include(GNUInstallDirs)
if(APPLE)
  install(TARGETS ClassicMenu BUNDLE DESTINATION .)
  set(CLOVER_RESOURCE_DEST "ClassicMenu.app/Contents/Resources")
else()
  install(TARGETS ClassicMenu RUNTIME DESTINATION .)
  set(CLOVER_RESOURCE_DEST ".")
endif()
if(EXISTS "${CLOVER_ASSET_DIR}/catalog.json")
  install(DIRECTORY "${CLOVER_ASSET_DIR}/" DESTINATION "${CLOVER_RESOURCE_DEST}/assets")
else()
  message(STATUS "Image-driven launcher: no Nintendo assets in package; import a local image at runtime")
endif()
install(FILES third_party/ares/LICENSE DESTINATION "${CLOVER_RESOURCE_DEST}/licenses" RENAME ares-LICENSE)
install(FILES third_party/SDL/LICENSE.txt DESTINATION "${CLOVER_RESOURCE_DEST}/licenses" RENAME SDL-LICENSE.txt)
install(FILES third_party/vendor/json-LICENSE.MIT third_party/vendor/stb-LICENSE dependencies.lock.json DESTINATION "${CLOVER_RESOURCE_DEST}/licenses")
install(FILES portable/core/LICENSE.ares DESTINATION "${CLOVER_RESOURCE_DEST}/licenses" RENAME ares-core-LICENSE)
install(FILES third_party/zlib/LICENSE DESTINATION "${CLOVER_RESOURCE_DEST}/licenses" RENAME zlib-LICENSE)
if(CLOVER_BUILD_TESTS)
  add_test(NAME clover_launcher_smoke COMMAND ClassicMenu --data-dir "${CMAKE_BINARY_DIR}/launcher-smoke-data" --render-launcher "${CMAKE_BINARY_DIR}/launcher-smoke.png")
  set_tests_properties(clover_launcher_smoke PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_AUDIODRIVER=dummy" TIMEOUT 30)
endif()
if(CLOVER_BUILD_TESTS AND EXISTS "${CLOVER_ASSET_DIR}/catalog.json")
  add_test(NAME clover_menu_smoke COMMAND ClassicMenu --assets "${CLOVER_ASSET_DIR}" --data-dir "${CMAKE_BINARY_DIR}/smoke-data" --smoke-test --screenshot "${CMAKE_BINARY_DIR}/menu-smoke.png")
  set_tests_properties(clover_menu_smoke PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_AUDIODRIVER=dummy" TIMEOUT 30)
endif()
