cmake_minimum_required(VERSION 3.28)
project(CloverAppleInputTests LANGUAGES CXX)
find_package(SDL3 REQUIRED CONFIG)
enable_testing()
add_executable(clover_apple_input_tests input_tests.cpp)
target_compile_features(clover_apple_input_tests PRIVATE cxx_std_20)
target_compile_options(clover_apple_input_tests PRIVATE -Wall -Wextra -Werror)
target_link_libraries(clover_apple_input_tests PRIVATE SDL3::SDL3-static)
add_test(NAME clover_apple_input_tests COMMAND clover_apple_input_tests)
set_tests_properties(clover_apple_input_tests PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_AUDIODRIVER=dummy")
add_executable(clover_apple_touch_tests touch_tests.cpp)
target_compile_features(clover_apple_touch_tests PRIVATE cxx_std_20)
target_compile_options(clover_apple_touch_tests PRIVATE -Wall -Wextra -Werror)
target_link_libraries(clover_apple_touch_tests PRIVATE SDL3::SDL3-static)
add_test(NAME clover_apple_touch_tests COMMAND clover_apple_touch_tests)
